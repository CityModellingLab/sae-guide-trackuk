Temporary Standardised Residual Maps

Model estimates compared with direct estimates

This temporary workbook gathers the existing direct, FH, BHF, and MRP outputs and computes standardised residuals in one place. These residuals compare each model estimate with the corresponding direct survey estimate, so they are useful diagnostics rather than ground-truth validation.

Code 
direct_estimates <- read_csv(
  file.path(base_path, "freqwalk_estimates_direct.csv"),
  show_col_types = FALSE
) |>
  select(area_id, time_id, msoa_name, direct_estimate, variance, n)

read_method_estimates <- function(path, method_name, estimate_col, variance_col) {
  read_csv(path, show_col_types = FALSE) |>
    transmute(
      method = method_name,
      area_id,
      time_id,
      model_estimate = .data[[estimate_col]],
      model_variance = .data[[variance_col]]
    )
}

method_estimates <- bind_rows(
  read_method_estimates(
    file.path(base_path, "freqwalk_estimates_fh.csv"),
    "FH",
    "fh_estimate",
    "fh_variance"
  ),
  read_method_estimates(
    file.path(base_path, "freqwalk_estimates_bhf.csv"),
    "BHF",
    "bhf_estimate",
    "bhf_variance"
  ),
  read_method_estimates(
    file.path(base_path, "freqwalk_estimates_mrp.csv"),
    "MRP",
    "mrp_estimate",
    "mrp_variance"
  )
)

standardised_residuals <- method_estimates |>
  left_join(direct_estimates, by = c("area_id", "time_id")) |>
  mutate(
    method = factor(method, levels = c("FH", "BHF", "MRP")),
    std_resid = (direct_estimate - model_estimate) /
      sqrt(variance + model_variance),
    abs_std_resid = abs(std_resid)
  )
Code 
standardised_residuals |>
  group_by(method) |>
  summarise(
    area_time_cells = n(),
    mean_abs_std_resid = mean(abs_std_resid, na.rm = TRUE),
    median_abs_std_resid = median(abs_std_resid, na.rm = TRUE),
    p90_abs_std_resid = quantile(abs_std_resid, 0.9, na.rm = TRUE, names = FALSE),
    cells_abs_gt_2 = sum(abs_std_resid > 2, na.rm = TRUE),
    .groups = "drop"
  ) |>
  mutate(across(where(is.numeric), ~ round(.x, 2)))
